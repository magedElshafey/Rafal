function geometry(root) {
  const outside = [...root.querySelectorAll('*')].filter(e => {
    if (!e.getClientRects().length || getComputedStyle(e).visibility === 'hidden' || e.classList.contains('sr-only') || e.closest('[aria-hidden=true]')) return false;
    if (e.closest('[aria-roledescription=carousel]')) return false;
    let parent = e.parentElement;
    while(parent && parent !== root) { if(['auto','scroll'].includes(getComputedStyle(parent).overflowX)) return false; parent = parent.parentElement; }
    const r=e.getBoundingClientRect(); return r.width>0 && (r.left < -1 || r.right > innerWidth+1);
  }).map(e=>({tag:e.tagName,text:e.textContent.slice(0,70),class:e.className}));
  return {dir:document.documentElement.dir,documentOverflow:document.documentElement.scrollWidth>innerWidth, outside,
    mainZ:getComputedStyle(root).zIndex,isolation:getComputedStyle(root).isolation,
    cartMarkers:root.querySelectorAll('[data-cart-page]').length,pdpMarkers:root.querySelectorAll('[data-product-page]').length};
}
function hitButton(e) {
  const r=e.getBoundingClientRect();
  return {rect:r.toJSON(),points:[r.top+4,r.top+r.height/2,r.bottom-4].map(y=>{const hit=document.elementFromPoint(r.left+r.width/2,y);return Boolean(hit && (hit===e||e.contains(hit)));})};
}
