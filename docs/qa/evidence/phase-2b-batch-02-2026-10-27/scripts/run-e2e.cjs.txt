const {spawn}=require('node:child_process');
const fs=require('node:fs');
const path=require('node:path');
const root='D:/megz/Rafal', dir=__dirname;
const mode=process.argv[2]||'dev';
const pidLog=path.join(dir,mode+'-pids.jsonl');
const logPath=path.join(dir,mode+'-e2e.log');
fs.writeFileSync(pidLog,'');
const log=fs.createWriteStream(logPath);
const args=['node_modules/@playwright/test/cli.js','test','--global-timeout=180000','--reporter=line','--output='+path.join(dir,mode+'-results')];
const env={...process.env,DEBUG:'pw:webserver,pw:browser',QA_PID_LOG:pidLog,NODE_OPTIONS:'--require='+path.join(dir,'pid-probe.cjs'),NEXT_TELEMETRY_DISABLED:'1'};
delete env.MAP_E2E_LIVE; delete env.MAP_E2E_NO_TOKEN;
if(mode==='production') env.MAP_E2E_PRODUCTION='1'; else delete env.MAP_E2E_PRODUCTION;
const child=spawn(process.execPath,args,{cwd:root,env,windowsHide:true,stdio:['ignore','pipe','pipe']});
const started=Date.now(); let timedOut=false;
for(const s of [child.stdout,child.stderr]) s.on('data',d=>log.write(d));
console.log(JSON.stringify({mode,pid:child.pid,command:[process.execPath,...args],limitSeconds:210,logPath}));
const timer=setTimeout(()=>{
 timedOut=true;
 console.log('OUTER TIMEOUT: cleaning only instrumented Node descendants of this run');
 const records=fs.readFileSync(pidLog,'utf8').trim().split('\n').filter(Boolean).map(JSON.parse);
 for(const r of records.reverse()) if(r.pid!==child.pid){try{process.kill(r.pid);}catch{}}
 try{child.kill();}catch{}
},210000);
child.on('close',(code,signal)=>{
 clearTimeout(timer);log.end();
 const result={mode,code,signal,timedOut,effectiveExitCode:timedOut?124:code,durationSeconds:(Date.now()-started)/1000};
 fs.writeFileSync(path.join(dir,mode+'-e2e-result.json'),JSON.stringify(result,null,2)); console.log(JSON.stringify(result));
 process.exitCode=timedOut?124:code;
});
